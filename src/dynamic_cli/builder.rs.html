1//! Fluent builder API for creating CLI/REPL applications
2//!
3//! This module provides a builder pattern for easily constructing
4//! CLI and REPL applications with minimal boilerplate.
5//!
6//! # Example
7//!
8//! ```no_run
9//! use dynamic_cli::prelude::*;
10//!
11//! // Define context
12//! #[derive(Default)]
13//! struct MyContext;
14//!
15//! impl ExecutionContext for MyContext {
16//!     fn as_any(&self) -> &dyn std::any::Any { self }
17//!     fn as_any_mut(&mut self) -> &mut dyn std::any::Any { self }
18//! }
19//!
20//! // Define handler
21//! struct HelloCommand;
22//!
23//! impl CommandHandler for HelloCommand {
24//!     fn execute(
25//!         &self,
26//!         _context: &mut dyn ExecutionContext,
27//!         args: &ParsedArgs,
28//!     ) -> dynamic_cli::Result<()> {
29//!         println!("Hello!");
30//!         Ok(())
31//!     }
32//! }
33//!
34//! # fn main() -> dynamic_cli::Result<()> {
35//! // Build and run
36//! CliBuilder::new()
37//!     .config_file("commands.yaml")
38//!     .context(Box::new(MyContext::default()))
39//!     .register_sync_handler("hello_handler", Box::new(HelloCommand))
40//!     .build()?
41//!     .run()
42//! # }
43//! ```

57/// Fluent builder for creating CLI/REPL applications
58///
59/// Provides a chainable API for configuring and building applications.
60/// Automatically loads configuration, registers handlers, and creates
61/// the appropriate interface (CLI or REPL).
62///
63/// # Builder Pattern
64///
65/// The builder follows the standard Rust builder pattern:
66/// - Methods consume `self` and return `Self`
67/// - Final `build()` method consumes the builder and returns the app
68///
69/// # Example
70///
71/// ```no_run
72/// use dynamic_cli::prelude::*;
73///
74/// # #[derive(Default)]
75/// # struct MyContext;
76/// # impl ExecutionContext for MyContext {
77/// #     fn as_any(&self) -> &dyn std::any::Any { self }
78/// #     fn as_any_mut(&mut self) -> &mut dyn std::any::Any { self }
79/// # }
80/// # struct MyHandler;
81/// # impl CommandHandler for MyHandler {
82/// #     fn execute(&self, _: &mut dyn ExecutionContext, _: &dynamic_cli::parser::ParsedArgs) -> dynamic_cli::Result<()> { Ok(()) }
83/// # }
84/// # fn main() -> dynamic_cli::Result<()> {
85/// let app = CliBuilder::new()
86///     .config_file("commands.yaml")
87///     .context(Box::new(MyContext::default()))
88///     .register_sync_handler("my_handler", Box::new(MyHandler))
89///     .prompt("myapp")
90///     .build()?;
91/// # Ok(())
92/// # }
93/// ```
94pub struct CliBuilder {
95    /// Path to configuration file
96    config_path: Option<PathBuf>,
97
98    /// Loaded configuration
99    config: Option<CommandsConfig>,
100
101    /// Execution context
102    context: Option<Box<dyn ExecutionContext>>,
103
104    /// Registered command handlers (name -> handler)
105    handlers: HashMap<String, Box<dyn CommandHandler>>,
106
107    /// Registered asynchronous command handler (name -> async_handler)
108    async_handlers: HashMap<String, Box<dyn AsyncCommandHandler>>,
109
110    /// Registered plugins, expanded into `handlers` during `build()`
111    plugins: Vec<Box<dyn Plugin>>,
112
113    /// REPL prompt (if None, will use config default or "cli")
114    prompt: Option<String>,
115
116    /// Base override for the REPL's `\`-continuation prompt (DD-027, #68).
117    /// Forwarded verbatim to [`ReplInterface::with_prompt_multiline`] when
118    /// set; `None` keeps the "..." default. No effect in CLI one-shot mode.
119    prompt_multiline: Option<String>,
120
121    /// Custom help formatter. None = DefaultHelpFormatter used lazily.
122    help_formatter: Option<Box<dyn HelpFormatter>>,
123}

230    /// Register a (sync) command handler
231    ///
232    /// Associates a handler with the command's implementation name from the config.
233    /// The name must match the `implementation` field in the command definition.
234    ///
235    /// Renamed from `register_handler()` in v0.5.0 for symmetry with
236    /// [`register_async_handler`][Self::register_async_handler].
237    /// `register_handler()` remains available as a deprecated alias until
238    /// v1.0.0 (DD-022).
239    ///
240    /// # Arguments
241    ///
242    /// * `name` - Implementation name from the configuration
243    /// * `handler` - Boxed command handler implementing `CommandHandler`
244    ///
245    /// # Example
246    ///
247    /// ```
248    /// use dynamic_cli::prelude::*;
249    ///
250    /// struct MyCommand;
251    ///
252    /// impl CommandHandler for MyCommand {
253    ///     fn execute(
254    ///         &self,
255    ///         _ctx: &mut dyn ExecutionContext,
256    ///         _args: &ParsedArgs,
257    ///     ) -> dynamic_cli::Result<()> {
258    ///         println!("Executed!");
259    ///         Ok(())
260    ///     }
261    /// }
262    ///
263    /// let builder = CliBuilder::new()
264    ///     .register_sync_handler("my_command", Box::new(MyCommand));
265    /// ```
266    pub fn register_sync_handler(
267        mut self,
268        name: impl Into<String>,
269        handler: Box<dyn CommandHandler>,
270    ) -> Self {
271        self.handlers.insert(name.into(), handler);
272        self
273    }

290    /// Register an async command handler (DD-022)
291    ///
292    /// Additive counterpart of [`register_sync_handler`][Self::register_sync_handler].
293    /// Associates an async handler with the command's implementation name
294    /// from the config, exactly like its sync counterpart. Registering both
295    /// a sync and an async handler for the *same* implementation name is
296    /// detected as an error at [`build()`][Self::build] time, not here —
297    /// this method, like `register_sync_handler`, always succeeds (it just
298    /// stores the handler for `build()` to consume).
299    ///
300    /// # Arguments
301    ///
302    /// * `name` - Implementation name from the configuration
303    /// * `handler` - Boxed async command handler implementing `AsyncCommandHandler`
304    ///
305    /// # Example
306    ///
307    /// ```
308    /// use dynamic_cli::prelude::*;
309    /// use dynamic_cli::executor::AsyncCommandHandler;
310    /// use async_trait::async_trait;
311    ///
312    /// struct FetchCommand;
313    ///
314    /// #[async_trait]
315    /// impl AsyncCommandHandler for FetchCommand {
316    ///     async fn execute(
317    ///         &self,
318    ///         _ctx: &mut dyn ExecutionContext,
319    ///         _args: &ParsedArgs,
320    ///     ) -> dynamic_cli::Result<()> {
321    ///         println!("Fetched!");
322    ///         Ok(())
323    ///     }
324    /// }
325    ///
326    /// let builder = CliBuilder::new()
327    ///     .register_async_handler("fetch_command", Box::new(FetchCommand));
328    /// ```
329    pub fn register_async_handler(
330        mut self,
331        name: impl Into<String>,
332        handler: Box<dyn AsyncCommandHandler>,
333    ) -> Self {
334        self.async_handlers.insert(name.into(), handler);
335        self
336    }

338    /// Register a plugin
339    ///
340    /// A plugin groups related handlers under a single unit. During
341    /// [`build()`][Self::build], each handler declared by the plugin is
342    /// merged into the handler map. Conflicts with already-registered
343    /// handler names (whether from another plugin or from
344    /// [`register_sync_handler`][Self::register_sync_handler]) produce a
345    /// build-time error.
346    ///
347    /// The YAML config remains the sole source of truth for command
348    /// definitions. Plugin handlers are matched by their `implementation`
349    /// name, exactly as with [`register_sync_handler`][Self::register_sync_handler].
350    /// Plugins remain sync-only for now — out of scope for DD-022.
351    ///
352    /// # Example
353    ///
354    /// ```
355    /// use dynamic_cli::CliBuilder;
356    /// use dynamic_cli::plugin::SystemPlugin;
357    ///
358    /// let builder = CliBuilder::new()
359    ///     .register_plugin(Box::new(SystemPlugin::new()));
360    /// ```
361    pub fn register_plugin(mut self, plugin: Box<dyn Plugin>) -> Self {
362        self.plugins.push(plugin);
363        self
364    }

366    /// Load a WASM plugin from disk, map its business functions, and
367    /// register it
368    ///
369    /// Convenience wrapper around [`WasmPlugin::load`][crate::plugin::wasm::WasmPlugin::load],
370    /// [`WasmPlugin::with_function_map`][crate::plugin::wasm::WasmPlugin::with_function_map],
371    /// and [`register_plugin`][Self::register_plugin]. Only available with
372    /// the `wasm-plugins` feature.
373    ///
374    /// `function_map` is mandatory here, unlike the other `WasmPlugin`
375    /// builder methods (`with_format`, `with_metadata`), which have
376    /// reasonable defaults (YAML, file-name-derived metadata). A
377    /// `WasmPlugin` with an empty function map registers zero handlers —
378    /// silently inert — so this wrapper does not offer a path that skips
379    /// mapping. Applications that also need a non-default format or
380    /// explicit metadata should build the `WasmPlugin` directly and pass it
381    /// to [`register_plugin`][Self::register_plugin] instead.
382    ///
383    /// # Errors
384    ///
385    /// Returns an error if the module cannot be loaded or fails mandatory
386    /// export validation. See `WASM_PLUGIN_INTERFACE.md` for the ABI
387    /// contract.
388    ///
389    /// # Example
390    ///
391    /// ```no_run
392    /// use dynamic_cli::CliBuilder;
393    /// use std::path::Path;
394    ///
395    /// # fn main() -> dynamic_cli::Result<()> {
396    /// let builder = CliBuilder::new()
397    ///     .register_wasm_plugin(
398    ///         Path::new("plugins/greet.wasm"),
399    ///         &[("greet_hello", "say_hello")],
400    ///     )?;
401    /// # Ok(())
402    /// # }
403    /// ```
404    #[cfg(feature = "wasm-plugins")]
405    pub fn register_wasm_plugin(
406        self,
407        path: &std::path::Path,
408        function_map: &[(&str, &str)],
409    ) -> Result<Self> {
410        let mut plugin = crate::plugin::wasm::WasmPlugin::load(path)?;
411        for &(impl_name, wasm_fn_name) in function_map {
412            plugin = plugin.with_function_map(impl_name, wasm_fn_name);
413        }
414        Ok(self.register_plugin(Box::new(plugin)))
415    }

508    /// Build the application
509    ///
510    /// Performs the following steps:
511    /// 1. Load configuration (if `config_file()` was used)
512    /// 2. Validate that a context was provided
513    /// 3. Create the command registry
514    /// 4. Register all command handlers
515    /// 5. Verify that all required commands have handlers
516    /// 6. Create the `CliApp`
517    ///
518    /// # Returns
519    ///
520    /// A configured `CliApp` ready to run
521    ///
522    /// # Errors
523    ///
524    /// - Configuration errors (file not found, invalid format, etc.)
525    /// - Missing context
526    /// - Missing required handlers
527    /// - Registry errors
528    ///
529    /// # Example
530    ///
531    /// ```no_run
532    /// use dynamic_cli::prelude::*;
533    ///
534    /// # #[derive(Default)]
535    /// # struct MyContext;
536    /// # impl ExecutionContext for MyContext {
537    /// #     fn as_any(&self) -> &dyn std::any::Any { self }
538    /// #     fn as_any_mut(&mut self) -> &mut dyn std::any::Any { self }
539    /// # }
540    /// # struct MyHandler;
541    /// # impl CommandHandler for MyHandler {
542    /// #     fn execute(&self, _: &mut dyn ExecutionContext, _: &dynamic_cli::parser::ParsedArgs) -> dynamic_cli::Result<()> { Ok(()) }
543    /// # }
544    /// # fn main() -> dynamic_cli::Result<()> {
545    /// let app = CliBuilder::new()
546    ///     .config_file("commands.yaml")
547    ///     .context(Box::new(MyContext::default()))
548    ///     .register_sync_handler("handler", Box::new(MyHandler))
549    ///     .build()?;
550    ///
551    /// // Now app is ready to run
552    /// # Ok(())
553    /// # }
554    /// ```
555    pub fn build(mut self) -> Result<CliApp> {
556        // Load configuration if path was specified
557        let config = if let Some(config) = self.config.take() {
558            config
559        } else if let Some(path) = self.config_path.take() {
560            load_config(path)?
561        } else {
562            return Err(DynamicCliError::Config(ConfigError::InvalidSchema {
563                reason: "No configuration provided. Use config_file() or config()".to_string(),
564                path: None,
565                suggestion: None,
566            }));
567        };
568
569        // Validate context was provided
570        let context = self.context.take().ok_or_else(|| {
571            DynamicCliError::Config(ConfigError::InvalidSchema {
572                reason: "No execution context provided. Use context()".to_string(),
573                path: None,
574                suggestion: None,
575            })
576        })?;
577
578        // Create registry and register commands
579        let mut registry = CommandRegistry::new();
580
581        // Expand plugins into the handler map before processing commands.
582        // Conflicts between plugins (or between a plugin and a directly
583        // registered handler) are detected here and produce a clear error,
584        // before any command resolution begins.
585        for plugin in self.plugins.drain(..) {
586            let plugin_name = plugin.name().to_string();
587            for (impl_name, handler) in plugin.handlers() {
588                if self.handlers.contains_key(&impl_name) {
589                    return Err(DynamicCliError::Config(ConfigError::InvalidSchema {
590                        reason: format!(
591                            "Plugin '{}' tried to register handler '{}' \
592                             which is already registered.",
593                            plugin_name, impl_name
594                        ),
595                        path: None,
596                        suggestion: Some(format!(
597                            "Remove the duplicate call to register_sync_handler(\"{}\") \
598                             or rename the implementation in your YAML config.",
599                            impl_name
600                        )),
601                    }));
602                }
603                self.handlers.insert(impl_name, handler);
604            }
605        }
606
607        for command_def in &config.commands {
608            // Find handlers for this command — sync and async are looked up
609            // independently; at most one should be present (see conflict
610            // check below).
611            let sync_handler = self.handlers.remove(&command_def.implementation);
612            let async_handler = self.async_handlers.remove(&command_def.implementation);
613
614            // A command's implementation name must resolve to exactly one
615            // handler kind. Registering both is a configuration mistake,
616            // not something to resolve silently (e.g. "sync wins") — that
617            // would hide a bug where the same name was registered twice
618            // with different handler types.
619            if sync_handler.is_some() && async_handler.is_some() {
620                return Err(DynamicCliError::Config(ConfigError::InvalidSchema {
621                    reason: format!(
622                        "Command '{}' has both a sync and an async handler \
623                         registered for implementation '{}'.",
624                        command_def.name, command_def.implementation
625                    ),
626                    path: None,
627                    suggestion: Some(
628                        "Use either register_sync_handler() or \
629                         register_async_handler() for this implementation, not both."
630                            .to_string(),
631                    ),
632                }));
633            }
634
635            // Check if handler is required
636            if command_def.required && sync_handler.is_none() && async_handler.is_none() {
637                return Err(DynamicCliError::Config(ConfigError::InvalidSchema {
638                    reason: format!(
639                        "Required command '{}' has no registered handler (implementation: '{}'). \
640                        Use register_sync_handler() or register_async_handler() to register it.",
641                        command_def.name, command_def.implementation
642                    ),
643                    path: None,
644                    suggestion: None,
645                }));
646            }
647
648            // Register command with whichever handler kind was found.
649            if let Some(handler) = sync_handler {
650                registry.register_sync(command_def.clone(), handler)?;
651            } else if let Some(handler) = async_handler {
652                registry.register_async(command_def.clone(), handler)?;
653            }
654        }
655
656        // Determine prompt
657        let prompt = self
658            .prompt
659            .or_else(|| Some(config.metadata.prompt.clone()))
660            .unwrap_or_else(|| "cli".to_string());
661
662        Ok(CliApp {
663            registry,
664            context,
665            prompt,
666            prompt_multiline: self.prompt_multiline,
667            config,
668            help_formatter: self.help_formatter,
669        })
670    }
671}

679/// Built CLI/REPL application
680///
681/// Created by `CliBuilder::build()`. Provides methods to run the application
682/// in different modes:
683/// - `run()` - Auto-detect CLI vs REPL based on arguments
684/// - `run_cli()` - Force CLI mode with specific arguments
685/// - `run_repl()` - Force REPL mode
686///
687/// # Example
688///
689/// ```no_run
690/// use dynamic_cli::prelude::*;
691///
692/// # #[derive(Default)]
693/// # struct MyContext;
694/// # impl ExecutionContext for MyContext {
695/// #     fn as_any(&self) -> &dyn std::any::Any { self }
696/// #     fn as_any_mut(&mut self) -> &mut dyn std::any::Any { self }
697/// # }
698/// # struct MyHandler;
699/// # impl CommandHandler for MyHandler {
700/// #     fn execute(&self, _: &mut dyn ExecutionContext, _: &dynamic_cli::parser::ParsedArgs) -> dynamic_cli::Result<()> { Ok(()) }
701/// # }
702/// # fn main() -> dynamic_cli::Result<()> {
703/// let app = CliBuilder::new()
704///     .config_file("commands.yaml")
705///     .context(Box::new(MyContext::default()))
706///     .register_sync_handler("handler", Box::new(MyHandler))
707///     .build()?;
708///
709/// // Auto-detect mode (CLI if args provided, REPL otherwise)
710/// app.run()
711/// # }
712/// ```
713pub struct CliApp {
714    /// Command registry
715    registry: CommandRegistry,
716
717    /// Execution context
718    context: Box<dyn ExecutionContext>,
719
720    /// REPL prompt
721    prompt: String,
722
723    /// Base override for the REPL's `\`-continuation prompt (DD-027, #68).
724    /// Forwarded to `ReplInterface::with_prompt_multiline` in `run_repl()`
725    /// when set. Unused outside REPL mode.
726    prompt_multiline: Option<String>,
727
728    /// Full configuration - needed by the help formatter
729    config: CommandsConfig,
730
731    /// Custom help formatter, or None to use DefaultHelpFormatter
732    help_formatter: Option<Box<dyn HelpFormatter>>,
733}
734
735impl std::fmt::Debug for CliApp {
736    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
737        f.debug_struct("CliApp")
738            .field("prompt", &self.prompt)
739            .field("prompt_multiline", &self.prompt_multiline)
740            .field("registry", &"<CommandRegistry>")
741            .field("context", &"<ExecutionContext>")
742            .field("help_formatter", &"<Option<Box<dyn HelpFormatter>>>")
743            .finish()
744    }
745}

747impl CliApp {
748    /// Run in CLI mode with provided arguments
749    ///
750    /// Executes a single command and exits.
751    ///
752    /// # Arguments
753    ///
754    /// * `args` - Command-line arguments (typically from `env::args().skip(1)`)
755    ///
756    /// # Returns
757    ///
758    /// - `Ok(())` on successful execution
759    /// - `Err(...)` on parse, validation, or execution errors
760    ///
761    /// # Example
762    ///
763    /// ```no_run
764    /// # use dynamic_cli::prelude::*;
765    /// # #[derive(Default)]
766    /// # struct MyContext;
767    /// # impl ExecutionContext for MyContext {
768    /// #     fn as_any(&self) -> &dyn std::any::Any { self }
769    /// #     fn as_any_mut(&mut self) -> &mut dyn std::any::Any { self }
770    /// # }
771    /// # struct MyHandler;
772    /// # impl CommandHandler for MyHandler {
773    /// #     fn execute(&self, _: &mut dyn ExecutionContext, _: &dynamic_cli::parser::ParsedArgs) -> dynamic_cli::Result<()> { Ok(()) }
774    /// # }
775    /// # fn main() -> dynamic_cli::Result<()> {
776    /// # let app = CliBuilder::new()
777    /// #     .config_file("commands.yaml")
778    /// #     .context(Box::new(MyContext::default()))
779    /// #     .register_sync_handler("handler", Box::new(MyHandler))
780    /// #     .build()?;
781    /// // Run with specific arguments
782    /// app.run_cli(vec!["command".to_string(), "arg1".to_string()])
783    /// # }
784    /// ```
785    pub fn run_cli(self, args: Vec<String>) -> Result<()> {
786        // Intercept --help before command dispatch.
787        // The formatter is instantiated lazily, only when --help is detected.
788        match args.as_slice() {
789            [flag] if flag == "--help" => {
790                let formatter: Box<dyn HelpFormatter> = self
791                    .help_formatter
792                    .unwrap_or_else(|| Box::new(DefaultHelpFormatter::new()));
793                print!("{}", formatter.format_app(&self.config));
794                return Ok(());
795            }
796            [flag, command] if flag == "--help" => {
797                let formatter: Box<dyn HelpFormatter> = self
798                    .help_formatter
799                    .unwrap_or_else(|| Box::new(DefaultHelpFormatter::new()));
800                print!("{}", formatter.format_command(&self.config, command));
801                return Ok(());
802            }
803            _ => {}
804        }
805
806        let cli = CliInterface::new(self.registry, self.context);
807        cli.run(args)
808    }
809
810    /// Run a batch of command lines read from a file (#41).
811    ///
812    /// Convenience wrapper around
813    /// [`CliInterface::run_script`][crate::interface::CliInterface::run_script]
814    /// — see there for the line format, tokenization, and error-policy
815    /// details.
816    ///
817    /// # Example
818    ///
819    /// ```no_run
820    /// # use dynamic_cli::prelude::*;
821    /// # #[derive(Default)]
822    /// # struct MyContext;
823    /// # impl ExecutionContext for MyContext {
824    /// #     fn as_any(&self) -> &dyn std::any::Any { self }
825    /// #     fn as_any_mut(&mut self) -> &mut dyn std::any::Any { self }
826    /// # }
827    /// # struct MyHandler;
828    /// # impl CommandHandler for MyHandler {
829    /// #     fn execute(&self, _: &mut dyn ExecutionContext, _: &dynamic_cli::parser::ParsedArgs) -> dynamic_cli::Result<()> { Ok(()) }
830    /// # }
831    /// # fn main() -> dynamic_cli::Result<()> {
832    /// # let app = CliBuilder::new()
833    /// #     .config_file("commands.yaml")
834    /// #     .context(Box::new(MyContext::default()))
835    /// #     .register_sync_handler("handler", Box::new(MyHandler))
836    /// #     .build()?;
837    /// let outcome = app.run_script("commands.txt", ScriptErrorPolicy::Continue)?;
838    /// println!("{}/{} lines succeeded", outcome.lines_succeeded, outcome.lines_executed);
839    /// # Ok(())
840    /// # }
841    /// ```
842    pub fn run_script(
843        self,
844        path: impl AsRef<std::path::Path>,
845        policy: crate::interface::ScriptErrorPolicy,
846    ) -> Result<crate::interface::ScriptOutcome> {
847        let cli = CliInterface::new(self.registry, self.context);
848        cli.run_script(path, policy)
849    }
850
851    /// Run in REPL mode
852    ///
853    /// Enters an interactive loop that continues until the user exits.
854    ///
855    /// # Returns
856    ///
857    /// - `Ok(())` when user exits normally
858    /// - `Err(...)` on critical errors (e.g., rustyline initialization failure)
859    ///
860    /// # Example
861    ///
862    /// ```no_run
863    /// # use dynamic_cli::prelude::*;
864    /// # #[derive(Default)]
865    /// # struct MyContext;
866    /// # impl ExecutionContext for MyContext {
867    /// #     fn as_any(&self) -> &dyn std::any::Any { self }
868    /// #     fn as_any_mut(&mut self) -> &mut dyn std::any::Any { self }
869    /// # }
870    /// # struct MyHandler;
871    /// # impl CommandHandler for MyHandler {
872    /// #     fn execute(&self, _: &mut dyn ExecutionContext, _: &dynamic_cli::parser::ParsedArgs) -> dynamic_cli::Result<()> { Ok(()) }
873    /// # }
874    /// # fn main() -> dynamic_cli::Result<()> {
875    /// # let app = CliBuilder::new()
876    /// #     .config_file("commands.yaml")
877    /// #     .context(Box::new(MyContext::default()))
878    /// #     .register_sync_handler("handler", Box::new(MyHandler))
879    /// #     .build()?;
880    /// // Start interactive REPL
881    /// app.run_repl()
882    /// # }
883    /// ```
884    pub fn run_repl(self) -> Result<()> {
885        let mut repl = ReplInterface::new(
886            self.registry,
887            self.context,
888            self.prompt,
889            Some(self.config),
890            self.help_formatter,
891        )?;
892
893        if let Some(prompt_multiline) = self.prompt_multiline {
894            repl = repl.with_prompt_multiline(prompt_multiline);
895        }
896
897        repl.run()
898    }

900    /// Run with automatic mode detection
901    ///
902    /// Decides between CLI and REPL based on command-line arguments:
903    /// - If arguments provided → CLI mode
904    /// - If no arguments → REPL mode
905    ///
906    /// This is the recommended method for most applications.
907    ///
908    /// # Returns
909    ///
910    /// - `Ok(())` on successful execution
911    /// - `Err(...)` on errors
912    ///
913    /// # Example
914    ///
915    /// ```no_run
916    /// # use dynamic_cli::prelude::*;
917    /// # #[derive(Default)]
918    /// # struct MyContext;
919    /// # impl ExecutionContext for MyContext {
920    /// #     fn as_any(&self) -> &dyn std::any::Any { self }
921    /// #     fn as_any_mut(&mut self) -> &mut dyn std::any::Any { self }
922    /// # }
923    /// # struct MyHandler;
924    /// # impl CommandHandler for MyHandler {
925    /// #     fn execute(&self, _: &mut dyn ExecutionContext, _: &dynamic_cli::parser::ParsedArgs) -> dynamic_cli::Result<()> { Ok(()) }
926    /// # }
927    /// # fn main() -> dynamic_cli::Result<()> {
928    /// # let app = CliBuilder::new()
929    /// #     .config_file("commands.yaml")
930    /// #     .context(Box::new(MyContext::default()))
931    /// #     .register_sync_handler("handler", Box::new(MyHandler))
932    /// #     .build()?;
933    /// // Auto-detect: CLI if args, REPL if no args
934    /// app.run()
935    /// # }
936    /// ```
937    pub fn run(self) -> Result<()> {
938        let args: Vec<String> = std::env::args().skip(1).collect();
939
940        if args.is_empty() {
941            // No arguments → REPL mode
942            self.run_repl()
943        } else {
944            // Arguments provided → CLI mode
945            self.run_cli(args)
946        }
947    }
